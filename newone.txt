pipeline {
    agent any
    tools {
        maven 'Maven'
    }
    stages {
        stage('Clone Repository') {
            steps {
                git branch: 'master', url: 'https://github.com/Msocial123/EcommerceApp.git'
            }
        }
        stage('Check Project Structure') {
            steps {
                sh 'ls -la'
            }
        }
        stage('Compile') {
            steps {
                dir('EcommerceApp') {  
                    sh 'mvn clean compile'
                }
            }
        } 
        stage('Test') {
            steps {
                dir('EcommerceApp') {
                    sh 'mvn test'
                }
            }
        }
        stage('SonarQube Analysis') {
            steps {
                dir('EcommerceApp') {
                    withSonarQubeEnv('SonarQube') {
                        sh 'mvn sonar:sonar'
                    }
                }
            }
        }
        stage('Package') {
            steps {
                dir('EcommerceApp') {
                    sh 'mvn package'
                }
            }
        }
}
}
